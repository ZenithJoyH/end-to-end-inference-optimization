执行命令（执行前固化）：

```bash
# PPU-07
docker commit --pause=true hy4-opt-20260914 hy4-preview-optimized:20261015-ppu13-clone

# PPU-07 -> PPU-13，由本地控制面串联
ssh PPU-07 'docker save hy4-preview-optimized:20261015-ppu13-clone | zstd -T0 -1' \
  | ssh PPU-13 'zstd -d | docker load'

# 直连管道失败后的 NFS 安全中转（PPU-07 后台）
docker save hy4-preview-optimized:20261015-ppu13-clone \
  | zstd -T0 -1 -c \
  > /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/images/hy4-preview-optimized-20261015-ppu13-clone.tar.zst.partial
mv ...tar.zst.partial ...tar.zst
sha256sum ...tar.zst > ...tar.zst.sha256

# 后台启动已保存脚本，避免堡垒机断链终止导出
nohup /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/commands/save-image.sh \
  > /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/logs/save-image.log 2>&1 &
echo $! > /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/manifests/save-image.pid

# PPU-13
zstd -d -c ...tar.zst | docker load

# PPU-13 后台导入已校验归档
nohup /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/commands/load-image.sh \
  > /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/logs/load-image.log 2>&1 &
echo $! > /mnt/nfs/users/jinghao/hy4-preview/optimize/20261015-hy4-preview-ppu13-clone/manifests/load-image.pid

# PPU-13
docker run -dit --init --name hy4-opt-20260914 \
  --privileged --network host --ipc host --security-opt label=disable \
  --device /dev/infiniband:/dev/infiniband:rwm \
  --device /dev/alixpu_ctl:/dev/alixpu_ctl:rwm \
  --device /dev/alixpu:/dev/alixpu:rwm \
  --ulimit memlock=-1:-1 --ulimit stack=67108864:67108864 \
  -v /mnt:/mnt \
  --workdir /mnt/nfs/users/jinghao \
  hy4-preview-optimized:20261015-ppu13-clone
```
